### Method Summary

**Anti-symmetry check** (two components):
1. Storage check: Verify no lower-triangular entry (i > j) appears in the raw SC table (the generator only stores upper-triangular pairs).
2. Self-bracket check: For all even generators X, verify `sc[(i,i)]` is empty (zero self-bracket).

Note: For odd generators X, the graded bracket gives `{X,X} = 2X²` which can be nonzero. The anti-symmetry relation `[X,X} = (-1)^{1·1} [X,X} = [X,X}` is a tautology for odd X — it places no constraint. The actual value was verified to be correct via the oscillator algebra in I03.

**Super Jacobi check**:

Verified the identity:
```
(-1)^{p(x)p(z)} [[x,y},z} + (-1)^{p(y)p(x)} [[y,z},x} + (-1)^{p(z)p(y)} [[z,x},y} = 0
```
for all ordered triples (x, y, z) with x ≤ y ≤ z (in PBW index order).

The composed bracket `[[X,Y},Z}` is evaluated as:
```
sum_V sc[X][Y][V] * sc[V][Z][W]  (coefficient of W)
```
working entirely in exact rational arithmetic (Fraction). Lower-triangular bracket `[j,i}` for j > i is derived from upper-triangular via:
```
sc[j][i][Z] = -(-1)^{p(i)*p(j)} * sc[i][j][Z]
```

### Implementation: `src/verify_C_structure.py`

```python
#!/usr/bin/env python3
"""
verify_C_structure.py

Load data/C_{n}_structure.json for n=1,2,3 and verify:
  1. Graded anti-symmetry: sc[Y][X][Z] = -(-1)^{p(X)*p(Y)} * sc[X][Y][Z]
  2. Super Jacobi identity (cyclic form):
     (-1)^{p(x)p(z)} [[x,y},z} + (-1)^{p(y)p(x)} [[y,z},x} + (-1)^{p(z)p(y)} [[z,x},y} = 0

Usage:
    python verify_C_structure.py [--data-dir <path>]
"""

import sys
import json
import argparse
from fractions import Fraction
from pathlib import Path
from collections import defaultdict


def load_schema(path):
    with open(path) as f:
        return json.load(f)


def build_bracket_table(schema):
    basis = schema['basis']
    labels = basis['even'] + basis['odd']
    parity = {lbl: schema['parity'][lbl] for lbl in labels}
    label_idx = {lbl: i for i, lbl in enumerate(labels)}
    sc_raw = defaultdict(lambda: defaultdict(Fraction))
    for entry in schema['structure_constants']:
        X, Y, Z, coeff = entry['X'], entry['Y'], entry['Z'], Fraction(entry['coeff'])
        i, j = label_idx[X], label_idx[Y]
        sc_raw[(i, j)][Z] += coeff
    return labels, parity, label_idx, sc_raw


def bracket_sc(i, j, labels, parity, label_idx, sc_raw):
    """Return [labels[i], labels[j]} as {label: Fraction}."""
    if i <= j:
        return dict(sc_raw.get((i, j), {}))
    else:
        px, py = parity[labels[j]], parity[labels[i]]
        sign = Fraction(-(-1) ** (px * py))
        raw = sc_raw.get((j, i), {})
        return {lbl: sign * c for lbl, c in raw.items() if c != 0}


def check_antisymmetry(labels, parity, label_idx, sc_raw):
    failures = []
    for (i, j) in sc_raw:
        if i > j:
            failures.append(f"  FAIL: sc_raw has lower-triangular entry ({labels[i]}, {labels[j]})")
    for i, lbl in enumerate(labels):
        if parity[lbl] == 0:
            self_br = sc_raw.get((i, i), {})
            if any(c != 0 for c in self_br.values()):
                failures.append(f"  FAIL self-bracket: [{lbl},{lbl}] = {dict(self_br)} (should be 0 for even)")
    return failures


def bracket_composed(i, j, k, labels, parity, label_idx, sc_raw):
    ij_br = bracket_sc(i, j, labels, parity, label_idx, sc_raw)
    result = defaultdict(Fraction)
    for lbl_v, c_ij in ij_br.items():
        v = label_idx[lbl_v]
        for lbl_w, c_vk in bracket_sc(v, k, labels, parity, label_idx, sc_raw).items():
            result[lbl_w] += c_ij * c_vk
    return {lbl: c for lbl, c in result.items() if c != 0}


def jacobi_residual(i, j, k, labels, parity, label_idx, sc_raw):
    px, py, pz = parity[labels[i]], parity[labels[j]], parity[labels[k]]
    s1, s2, s3 = Fraction((-1)**(px*pz)), Fraction((-1)**(py*px)), Fraction((-1)**(pz*py))
    t1 = bracket_composed(i, j, k, labels, parity, label_idx, sc_raw)
    t2 = bracket_composed(j, k, i, labels, parity, label_idx, sc_raw)
    t3 = bracket_composed(k, i, j, labels, parity, label_idx, sc_raw)
    residual = defaultdict(Fraction)
    for lbl, c in t1.items(): residual[lbl] += s1 * c
    for lbl, c in t2.items(): residual[lbl] += s2 * c
    for lbl, c in t3.items(): residual[lbl] += s3 * c
    return {lbl: c for lbl, c in residual.items() if c != 0}


def check_jacobi(labels, parity, label_idx, sc_raw):
    n_labels = len(labels)
    failures, total = [], 0
    for i in range(n_labels):
        for j in range(i, n_labels):
            for k in range(j, n_labels):
                total += 1
                res = jacobi_residual(i, j, k, labels, parity, label_idx, sc_raw)
                if res:
                    failures.append(
                        f"  FAIL Jacobi: ({labels[i]}, {labels[j]}, {labels[k]}) residual={res}"
                    )
    return total, failures


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--data-dir', default=None)
    args = parser.parse_args()
    data_dir = Path(args.data_dir) if args.data_dir else Path(__file__).parent.parent / 'data'

    print('=' * 60)
    print('C(n+1) Structure Constant Verification')
    print('=' * 60)
    overall_pass = True

    for n in [1, 2, 3]:
        fname = data_dir / f'C_{n}_structure.json'
        print(f'\n--- C({n+1}) = osp(2|{2*n}), n={n} [{fname.name}] ---')
        schema = load_schema(fname)
        labels, parity, label_idx, sc_raw = build_bracket_table(schema)
        dim_e = sum(1 for lbl in labels if parity[lbl] == 0)
        dim_o = sum(1 for lbl in labels if parity[lbl] == 1)
        print(f'  Loaded: dim {dim_e}|{dim_o}, {sum(len(v) for v in sc_raw.values())} SC entries')

        asym_fails = check_antisymmetry(labels, parity, label_idx, sc_raw)
        if asym_fails:
            print(f'  Anti-symmetry: FAIL')
            for f in asym_fails: print(f)
            overall_pass = False
        else:
            print(f'  Anti-symmetry: PASS ({len(sc_raw)} upper-triangular pairs, all even self-brackets zero)')

        print(f'  Super Jacobi: checking all triples...', end='', flush=True)
        total, jac_fails = check_jacobi(labels, parity, label_idx, sc_raw)
        if jac_fails:
            print(f' FAIL ({len(jac_fails)}/{total} triples failed)')
            for f in jac_fails[:10]: print(f)
            overall_pass = False
        else:
            print(f' PASS ({total} triples checked)')

    print('\n' + '=' * 60)
    print('OVERALL: ALL CHECKS PASSED' if overall_pass else 'OVERALL: SOME CHECKS FAILED')
    print('=' * 60)
    return 0 if overall_pass else 1


if __name__ == '__main__':
    sys.exit(main())
```
